# Mechanism: raise a country's AI deployment level (country scope).
# A level is one agi_tier_N modifier. It applies to AI and player countries
# alike and needs no technology, so it works at any start year. Levels only go up.

agi_clear_tier_modifiers = {
	if = { limit = { has_modifier = agi_tier_1 } remove_modifier = agi_tier_1 }
	if = { limit = { has_modifier = agi_tier_2 } remove_modifier = agi_tier_2 }
	if = { limit = { has_modifier = agi_tier_3 } remove_modifier = agi_tier_3 }
	if = { limit = { has_modifier = agi_tier_4 } remove_modifier = agi_tier_4 }
	if = { limit = { has_modifier = agi_tier_5 } remove_modifier = agi_tier_5 }
}

agi_set_tier_1 = {
	if = {
		limit = { NOT = { has_variable = agi_tier } }
		add_modifier = { name = agi_tier_1 }
		set_variable = { name = agi_tier value = 1 }
		if = { limit = { is_player = yes } trigger_event = { id = agi.1 } }
	}
}

agi_set_tier_2 = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = agi_tier }
				var:agi_tier < 2
			}
		}
		agi_clear_tier_modifiers = yes
		add_modifier = { name = agi_tier_2 }
		set_variable = { name = agi_tier value = 2 }
		if = { limit = { is_player = yes } trigger_event = { id = agi.2 } }
	}
}

agi_set_dividend_1 = {
	if = {
		limit = {
			NOT = { has_modifier = agi_dividend_1 }
			NOT = { has_modifier = agi_dividend_2 }
		}
		add_modifier = { name = agi_dividend_1 }
	}
}

agi_set_dividend_2 = {
	if = {
		limit = { NOT = { has_modifier = agi_dividend_2 } }
		if = { limit = { has_modifier = agi_dividend_1 } remove_modifier = agi_dividend_1 }
		add_modifier = { name = agi_dividend_2 }
	}
}

agi_notify_players = {
	every_country = {
		limit = { is_player = yes }
		if = {
			limit = { has_global_variable = agi_scenario_over }
			trigger_event = { id = agi.11 }
		}
		else_if = {
			limit = { has_game_rule = agi_scenario_ai2027_race }
			trigger_event = { id = agi.21 }
		}
		else_if = {
			limit = { has_game_rule = agi_scenario_ai2027_slowdown }
			trigger_event = { id = agi.22 }
		}
		else_if = {
			limit = { has_game_rule = agi_scenario_ai2040_plan_a }
			trigger_event = { id = agi.23 }
		}
	}
}

# Global scope, monthly. Starts the scenario clock at the chosen shock date,
# advances it one month per month, and applies the active scenario's schedule.
# It keeps running after the scenario ends, so a country CWE creates later
# (decolonisation, the Soviet break-up) still gets its group's final level.
agi_scenario_tick = {
	if = {
		limit = {
			NOT = { has_game_rule = agi_scenario_none }
			OR = {
				AND = { has_game_rule = agi_shock_start_1955 game_date >= 1955.1.1 }
				AND = { has_game_rule = agi_shock_start_1970 game_date >= 1970.1.1 }
				AND = { has_game_rule = agi_shock_start_1985 game_date >= 1985.1.1 }
			}
		}
		if = {
			limit = { NOT = { has_global_variable = agi_month } }
			set_global_variable = { name = agi_month value = 0 }
			agi_notify_players = yes
		}
		else = {
			change_global_variable = { name = agi_month add = 1 }
		}
		if = {
			limit = { has_game_rule = agi_scenario_ai2027_race }
			agi_schedule_ai2027_race = yes
		}
		else_if = {
			limit = { has_game_rule = agi_scenario_ai2027_slowdown }
			agi_schedule_ai2027_slowdown = yes
		}
		else_if = {
			limit = { has_game_rule = agi_scenario_ai2040_plan_a }
			agi_schedule_ai2040_plan_a = yes
		}
	}
}

# Fires once. Every schedule ends by calling it.
agi_end_scenario = {
	if = {
		limit = { NOT = { has_global_variable = agi_scenario_over } }
		set_global_variable = agi_scenario_over
		agi_notify_players = yes
	}
}
