# Scenario schedules (global scope). Month 0 = the shock start = January 2026 in
# the source scenario. Steps use ">=" so a missed month still applies. The tier
# and dividend effects only ever raise a level, so re-running a step is a no-op.
# docs/scenarios.md gives the source for each step and marks what is inferred.

# AI 2027, race ending (ai-2027.com). Stops in January 2030, before the
# scenario's takeover and mid-2030 extinction: an economic model has nothing to
# say about those.
agi_schedule_ai2027_race = {
	# Jan 2026: Agent-1 era agents at the two leading labs.
	if = { limit = { exists = c:USA } c:USA = { agi_set_tier_1 = yes } }
	every_country = { limit = { agi_is_china = yes } agi_set_tier_1 = yes }
	# Jul 2027: Agent-3-mini released publicly (cheap remote worker).
	if = {
		limit = { global_var:agi_month >= 18 exists = c:USA }
		c:USA = { agi_set_tier_2 = yes }
	}
	# Sep 2027: China about two months behind.
	if = {
		limit = { global_var:agi_month >= 20 }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_2 = yes }
	}
	# Dec 2027: Agent-5-designed software spreads through US business; GDP balloons.
	if = {
		limit = { global_var:agi_month >= 23 exists = c:USA }
		c:USA = { agi_set_tier_3 = yes }
	}
	# Jan 2028: the rest of the world gets cheap remote workers.
	if = {
		limit = { global_var:agi_month >= 24 }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_2 = yes }
	}
	# Mid 2028: Agent-5 released publicly; special economic zones and a robot build-out.
	if = {
		limit = { global_var:agi_month >= 30 }
		if = { limit = { exists = c:USA } c:USA = { agi_set_tier_4 = yes } }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_3 = yes }
	}
	# 2029: robots at scale in both powers; a universal basic income.
	if = {
		limit = { global_var:agi_month >= 36 }
		if = {
			limit = { exists = c:USA }
			c:USA = {
				agi_set_tier_5 = yes
				agi_set_dividend_1 = yes
			}
		}
		every_country = {
			limit = { agi_is_china = yes }
			agi_set_tier_4 = yes
			agi_set_dividend_1 = yes
		}
	}
	# Late 2029: special economic zones spread around the world.
	if = {
		limit = { global_var:agi_month >= 45 }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_3 = yes }
	}
	if = { limit = { global_var:agi_month >= 48 } agi_end_scenario = yes }
}

# AI 2027, slowdown ending (ai-2027.com/slowdown). Same as the race until the
# October 2027 branch point. AI research slows, but AI reaches the economy as
# fast as in the race and spreads further.
agi_schedule_ai2027_slowdown = {
	if = { limit = { exists = c:USA } c:USA = { agi_set_tier_1 = yes } }
	every_country = { limit = { agi_is_china = yes } agi_set_tier_1 = yes }
	if = {
		limit = { global_var:agi_month >= 18 exists = c:USA }
		c:USA = { agi_set_tier_2 = yes }
	}
	if = {
		limit = { global_var:agi_month >= 20 }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_2 = yes }
	}
	if = {
		limit = { global_var:agi_month >= 24 }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_2 = yes }
	}
	# May 2028: superhuman AI released to the public. The special economic zones
	# both powers announced in February are running.
	if = {
		limit = { global_var:agi_month >= 28 }
		if = { limit = { exists = c:USA } c:USA = { agi_set_tier_4 = yes } }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_4 = yes }
	}
	# Jun 2028: factories around the world run round the clock.
	if = {
		limit = { global_var:agi_month >= 29 }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_3 = yes }
	}
	# 2029: robots commonplace; a basic income and foreign aid end poverty,
	# even in developing countries.
	if = {
		limit = { global_var:agi_month >= 36 }
		if = { limit = { exists = c:USA } c:USA = { agi_set_tier_5 = yes } }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_4 = yes }
		every_country = {
			limit = { NOT = { is_country_type = decentralized } }
			agi_set_dividend_1 = yes
		}
	}
	# 2030: the transformation reaches everyone.
	if = {
		limit = { global_var:agi_month >= 48 }
		every_country = {
			limit = { NOT = { is_country_type = decentralized } }
			agi_set_tier_5 = yes
		}
	}
	if = { limit = { global_var:agi_month >= 60 } agi_end_scenario = yes }
}

# AI 2040: Plan A (ai-2040.com). A 2029 US-China deal pauses training and
# verifies compute. From 2032 robot and compute growth is capped and a citizen's
# dividend is paid. AI is held at top-expert level from 2035. The site's data
# puts China about two years behind the US and the rest of the world about three.
agi_schedule_ai2040_plan_a = {
	# 2027: millions of AI agents in the US economy.
	if = {
		limit = { global_var:agi_month >= 12 }
		if = { limit = { exists = c:USA } c:USA = { agi_set_tier_1 = yes } }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_1 = yes }
	}
	if = {
		limit = { global_var:agi_month >= 24 }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_1 = yes }
	}
	# 2029-2030: training pause and verification. Deployment holds (no step).
	# 2031: first regulated models; AI does a third of US cognitive labour.
	if = {
		limit = { global_var:agi_month >= 60 exists = c:USA }
		c:USA = { agi_set_tier_2 = yes }
	}
	# 2032: US real GDP up about 50%; caps on robots and compute; the dividend begins.
	if = {
		limit = { global_var:agi_month >= 72 }
		if = {
			limit = { exists = c:USA }
			c:USA = {
				agi_set_tier_3 = yes
				agi_set_dividend_1 = yes
			}
		}
		every_country = { limit = { agi_is_china = yes } agi_set_dividend_1 = yes }
	}
	if = {
		limit = { global_var:agi_month >= 84 }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_2 = yes }
	}
	if = {
		limit = { global_var:agi_month >= 96 }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_3 = yes }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_2 = yes }
	}
	# 2035: pause at top-expert AI; AI and robots do about 85% of US labour.
	# The US dividend is about $1M; foreign aid reaches about $10K per adult.
	if = {
		limit = { global_var:agi_month >= 108 }
		if = {
			limit = { exists = c:USA }
			c:USA = {
				agi_set_tier_4 = yes
				agi_set_dividend_2 = yes
			}
		}
		every_country = { limit = { agi_is_china = yes } agi_set_dividend_2 = yes }
		every_country = {
			limit = { agi_is_rest_of_world = yes }
			agi_set_tier_3 = yes
			agi_set_dividend_1 = yes
		}
	}
	# 2036: top-expert AI at full scale: 2 billion robots, a quarter of
	# Americans employed. Scaling to superintelligence only resumes in 2040.
	if = {
		limit = { global_var:agi_month >= 120 exists = c:USA }
		c:USA = { agi_set_tier_5 = yes }
	}
	if = {
		limit = { global_var:agi_month >= 132 }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_4 = yes }
	}
	if = {
		limit = { global_var:agi_month >= 144 }
		every_country = { limit = { agi_is_china = yes } agi_set_tier_5 = yes }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_4 = yes }
	}
	if = {
		limit = { global_var:agi_month >= 156 }
		every_country = { limit = { agi_is_rest_of_world = yes } agi_set_tier_5 = yes }
	}
	if = { limit = { global_var:agi_month >= 180 } agi_end_scenario = yes }
}
